- [ ] **Step 1: `meshcmp.cpp`**

```cpp
// meshcmp.cpp -- defect 012 bench: mesh ONE shape exactly like FreeCAD 1.1.1
// (ViewProviderPartExt::setupCoinGeometry, src/Mod/Part/Gui/ViewProviderExt.cpp:1096-1110 @0108fd4) and print
// time, peak memory and two digests of the result:
//   strict - FNV-1a over nodes, UV nodes, triangles and polygons on triangulation IN STORED ORDER;
//   canon  - order-independent: sorted node coordinates + triangles as coordinate triples (rotated so the
//            smallest vertex comes first, orientation kept), sorted.
// Usage: meshcmp.exe <brep> <angle_deg> <inparallel 0|1> <out.tsv> [deviation=0.5]
// A Job Object caps the process at 3 GB of committed memory (a prototype once took 5 GB).
#include <BRepBndLib.hxx>
#include <BRepMesh_IncrementalMesh.hxx>
#include <BRepTools.hxx>
#include <BRep_Builder.hxx>
#include <BRep_Tool.hxx>
#include <Bnd_Box.hxx>
#include <IMeshTools_Parameters.hxx>
#include <Poly_PolygonOnTriangulation.hxx>
#include <Poly_Triangulation.hxx>
#include <TopExp_Explorer.hxx>
#include <TopoDS.hxx>
#include <windows.h>
#include <psapi.h>
#include <algorithm>
#include <array>
#include <chrono>
#include <cmath>
#include <cstdint>
#include <cstdio>
#include <cstring>
#include <vector>

static void capMemory(SIZE_T bytes)
{
  HANDLE job = CreateJobObjectW(nullptr, nullptr);
  JOBOBJECT_EXTENDED_LIMIT_INFORMATION li = {};
  li.BasicLimitInformation.LimitFlags = JOB_OBJECT_LIMIT_PROCESS_MEMORY;
  li.ProcessMemoryLimit = bytes;
  SetInformationJobObject(job, JobObjectExtendedLimitInformation, &li, sizeof(li));
  AssignProcessToJobObject(job, GetCurrentProcess());
}

struct Fnv
{
  uint64_t h = 1469598103934665603ULL;
  void add(const void* p, size_t n)
  {
    const unsigned char* c = static_cast<const unsigned char*>(p);
    for (size_t i = 0; i < n; ++i) { h ^= c[i]; h *= 1099511628211ULL; }
  }
  template <class T> void val(const T& v) { add(&v, sizeof(v)); }
};

using P3 = std::array<double, 3>;

int main(int argc, char** argv)
{
  if (argc < 5) { std::fprintf(stderr, "usage: meshcmp <brep> <angle> <par> <out.tsv> [dev]\n"); return 2; }
  capMemory((SIZE_T)3 << 30);
  const char* path = argv[1];
  const double angle = std::atof(argv[2]);
  const bool par = std::atoi(argv[3]) != 0;
  const double dev = argc > 5 ? std::atof(argv[5]) : 0.5;

  TopoDS_Shape shape; BRep_Builder bb;
  if (!BRepTools::Read(shape, path, bb) || shape.IsNull()) { std::fprintf(stderr, "cannot read %s\n", path); return 3; }
  // Part::Tools::getDeflection (src/Mod/Part/App/Tools.cpp @0108fd4)
  Bnd_Box box; BRepBndLib::Add(shape, box); box.SetGap(0.0);
  double x0, y0, z0, x1, y1, z1; box.Get(x0, y0, z0, x1, y1, z1);
  double defl = ((x1 - x0) + (y1 - y0) + (z1 - z0)) / 300.0 * dev;
  if (defl < gp::Resolution()) defl = Precision::Confusion();

  IMeshTools_Parameters p;
  p.Deflection = defl; p.Relative = Standard_False; p.Angle = angle * M_PI / 180.0;
  p.InParallel = par ? Standard_True : Standard_False; p.AllowQualityDecrease = Standard_True;
  BRepTools::Clean(shape, Standard_True);
  const auto t0 = std::chrono::steady_clock::now();
  BRepMesh_IncrementalMesh mesher(shape, p);
  const double secs = std::chrono::duration<double>(std::chrono::steady_clock::now() - t0).count();

  Fnv strictAll, canonAll;
  long tri = 0, nodes = 0;
  std::string facesOut;
  int faceNo = 0;
  for (TopExp_Explorer fe(shape, TopAbs_FACE); fe.More(); fe.Next())
  {
    ++faceNo;
    const TopoDS_Face& F = TopoDS::Face(fe.Current());
    TopLoc_Location loc;
    Handle(Poly_Triangulation) T = BRep_Tool::Triangulation(F, loc);
    Fnv s, c;
    if (!T.IsNull())
    {
      tri += T->NbTriangles(); nodes += T->NbNodes();
      std::vector<P3> pts(T->NbNodes() + 1);
      for (int i = 1; i <= T->NbNodes(); ++i)
      {
        const gp_Pnt q = T->Node(i);
        pts[i] = {q.X(), q.Y(), q.Z()};
        s.val(pts[i]);
        if (T->HasUVNodes()) { const gp_Pnt2d u = T->UVNode(i); s.val(u.X()); s.val(u.Y()); }
      }
      std::vector<std::array<P3, 3>> tris;
      tris.reserve(T->NbTriangles());
      for (int i = 1; i <= T->NbTriangles(); ++i)
      {
        int a, b, d; T->Triangle(i).Get(a, b, d);
        s.val(a); s.val(b); s.val(d);
        std::array<P3, 3> t = {pts[a], pts[b], pts[d]};
        int m = 0;                                   // rotate: smallest vertex first, orientation kept
        for (int k = 1; k < 3; ++k) if (t[k] < t[m]) m = k;
        std::rotate(t.begin(), t.begin() + m, t.end());
        tris.push_back(t);
      }
      for (TopExp_Explorer ee(F, TopAbs_EDGE); ee.More(); ee.Next())
      {
        Handle(Poly_PolygonOnTriangulation) PT = BRep_Tool::PolygonOnTriangulation(TopoDS::Edge(ee.Current()), T, loc);
        if (PT.IsNull()) { s.val(-1); continue; }
        for (int i = 1; i <= PT->NbNodes(); ++i) s.val(PT->Node(i));
      }
      std::vector<P3> sortedPts(pts.begin() + 1, pts.end());
      std::sort(sortedPts.begin(), sortedPts.end());
      for (const P3& q : sortedPts) c.val(q);
      std::sort(tris.begin(), tris.end());
      for (const auto& t : tris) c.val(t);
    }
    strictAll.val(s.h); canonAll.val(c.h);
    char line[512];
    std::snprintf(line, sizeof(line), "%s\t%.2f\t%d\t%d\t%016llx\t%016llx\n", path, angle, par ? 1 : 0, faceNo,
                  (unsigned long long)s.h, (unsigned long long)c.h);
    facesOut += line;
  }
  PROCESS_MEMORY_COUNTERS pmc = {}; GetProcessMemoryInfo(GetCurrentProcess(), &pmc, sizeof(pmc));
  FILE* f = std::fopen(argv[4], "a");
  std::fprintf(f, "%s\t%.2f\t%d\t%.4f\t%ld\t%ld\t%.1f\t%016llx\t%016llx\n", path, angle, par ? 1 : 0, secs, tri, nodes,
               pmc.PeakWorkingSetSize / 1048576.0, (unsigned long long)strictAll.h, (unsigned long long)canonAll.h);
  std::fclose(f);
  std::string facesPath = std::string(argv[4]) + ".faces";
  FILE* ff = std::fopen(facesPath.c_str(), "a"); std::fputs(facesOut.c_str(), ff); std::fclose(ff);
  return 0;
}
```

```python
"""Defect 012 series: shapes x angles x DLL variants, interleaved per shape (stock, control, p012, stock, ...),
one meshcmp.exe process per (shape, angle, variant, rep); the variant is chosen by PATH (the DLL dir first).

python run_cmp.py --list LIST --angles 28.5,6.4 --par 0 --variants stock=C:/Program Files/FreeCAD 1.1/bin,...
                  --reps 1 --out DIR [--timeout 900] [--jobs 1]
"""
import argparse
import ctypes
import os
import subprocess

EXE = "C:/dev/freecad-kernel-fixes/build/meshcmp/meshcmp.exe"
FC_BIN = "C:/Program Files/FreeCAD 1.1/bin"


class MS(ctypes.Structure):
    _fields_ = [("dwLength", ctypes.c_ulong), ("dwMemoryLoad", ctypes.c_ulong), ("ullTotalPhys", ctypes.c_ulonglong),
                ("ullAvailPhys", ctypes.c_ulonglong), ("a", ctypes.c_ulonglong), ("b", ctypes.c_ulonglong),
                ("c", ctypes.c_ulonglong), ("d", ctypes.c_ulonglong), ("e", ctypes.c_ulonglong)]


def free_gb():
    m = MS(); m.dwLength = ctypes.sizeof(m)
    ctypes.windll.kernel32.GlobalMemoryStatusEx(ctypes.byref(m))
    return m.ullAvailPhys / 2 ** 30


def main():
    a = argparse.ArgumentParser()
    a.add_argument("--list"); a.add_argument("--angles"); a.add_argument("--par", default="0")
    a.add_argument("--variants"); a.add_argument("--reps", type=int, default=1); a.add_argument("--out")
    a.add_argument("--timeout", type=int, default=900)
    o = a.parse_args()
    os.makedirs(o.out, exist_ok=True)
    variants = [v.split("=", 1) for v in o.variants.split(",")]
    shapes = [s.strip() for s in open(o.list) if s.strip()]
    log = open(os.path.join(o.out, "run.log"), "a")
    for shape in shapes:
        for ang in o.angles.split(","):
            for rep in range(o.reps):
                for name, dll_dir in variants:
                    if free_gb() < 4.0:
                        log.write("SKIP lowmem %s %s %s\n" % (shape, ang, name)); log.flush()
                        continue
                    env = dict(os.environ)
                    env["PATH"] = dll_dir.replace("/", "\\") + ";" + FC_BIN.replace("/", "\\") + ";" + env["PATH"]
                    tsv = os.path.join(o.out, "%s.tsv" % name)
                    try:
                        r = subprocess.run([EXE, shape, ang, o.par, tsv], env=env, timeout=o.timeout,
                                           capture_output=True, text=True)
                        log.write("%s rc=%d %s %s %s\n" % (name, r.returncode, shape, ang, r.stderr.strip()[:200]))
                    except subprocess.TimeoutExpired:
                        log.write("TIMEOUT %s %s %s\n" % (name, shape, ang))
                    log.flush()


if __name__ == "__main__":
    main()
```

```python
"""Compare two meshcmp variants: identity (strict / canonical per shape and per face), speed, peak memory.
python compare.py DIR base variant [out.md]"""
import collections
import statistics
import sys


def load(path):
    rows = collections.defaultdict(list)
    for line in open(path, encoding="utf-8"):
        f = line.rstrip("\n").split("\t")
        if len(f) < 9:
            continue
        rows[(f[0], f[1], f[2])].append({"s": float(f[3]), "tri": int(f[4]), "nodes": int(f[5]),
                                          "mb": float(f[6]), "strict": f[7], "canon": f[8]})
    return rows


def main():
    d, base, var = sys.argv[1], sys.argv[2], sys.argv[3]
    A, B = load("%s/%s.tsv" % (d, base)), load("%s/%s.tsv" % (d, var))
    keys = sorted(set(A) & set(B))
    lines = ["| форма | угол | тр. %s | тр. %s | strict | canon | с %s | с %s | × | МБ %s | МБ %s |" % (base, var, base, var, base, var),
             "|---|---|---|---|---|---|---|---|---|---|---|"]
    n_strict = n_canon = 0
    worst_slow = 0.0
    for k in keys:
        a, b = A[k], B[k]
        sa, sb = statistics.median(r["s"] for r in a), statistics.median(r["s"] for r in b)
        st = all(r["strict"] == a[0]["strict"] for r in b)
        ca = all(r["canon"] == a[0]["canon"] for r in b)
        n_strict += st; n_canon += ca
        ratio = sa / sb if sb > 0 else float("inf")
        if sa > 0.05:
            worst_slow = max(worst_slow, sb / sa)
        lines.append("| %s | %s | %d | %d | %s | %s | %.3f | %.3f | %.2f× | %.0f | %.0f |" % (
            k[0].split("/")[-1], k[1], a[0]["tri"], b[0]["tri"], "=" if st else "≠", "=" if ca else "≠",
            sa, sb, ratio, max(r["mb"] for r in a), max(r["mb"] for r in b)))
    lines.append("")
    lines.append("формо-углов: %d; строго равны: %d; канонически равны: %d; худшее замедление (формы >50 мс): %.2f×"
                 % (len(keys), n_strict, n_canon, worst_slow))
    md = "\n".join(lines)
    if len(sys.argv) > 4:
        open(sys.argv[4], "w", encoding="utf-8").write(md + "\n")
    print(md)


if __name__ == "__main__":
    main()
```

```cpp
// Copyright (c) 2026 — modification of Open CASCADE Technology 7.8.1 for FreeCAD (defect 012,
// https://github.com/fozzfut/freecad-kernel-fixes). Distributed under GNU LGPL 2.1 with the Open CASCADE
// exception, as the rest of OCCT.
//
// Point-in-polygon with EXACTLY the answers of CSLib_Class2d::SiDans (OCCT 7.8.1, CSLib_Class2d.cxx: Init,
// SiDans, InternalSiDans, InternalSiDansOuOn, Transform2d) — every per-segment predicate uses the same
// floating-point expressions — but a query visits only the segments/vertices that can change its answer,
// found through uniform buckets over the normalized polygon:
//   parity (ray to +x)  : segments whose y-interval contains Py   -> y-buckets of segments
//   ON test 1 (vertex)  : vertices within the open box of Tolu/Tolv -> y-buckets of vertices
//   ON test 2 (x-cross) : segments whose x-interval strictly contains Px -> x-buckets of segments
// CSLib_Class2d scans all N segments for every query; on the owner's ball-screw face (14 149 boundary
// segments) that scan was 35-39 % of the meshing time (C:/dev/cad-perf-research/tessellation).
#ifndef _BRepMesh_BucketClass2d_HeaderFile
#define _BRepMesh_BucketClass2d_HeaderFile

#include <Standard.hxx>
#include <TColgp_Array1OfPnt2d.hxx>
#include <gp_Pnt2d.hxx>

#include <algorithm>
#include <vector>

class BRepMesh_BucketClass2d
{
public:
  DEFINE_STANDARD_ALLOC

  BRepMesh_BucketClass2d(const TColgp_Array1OfPnt2d& thePnts, const Standard_Real theTolU,
                         const Standard_Real theTolV, const Standard_Real theUMin, const Standard_Real theVMin,
                         const Standard_Real theUMax, const Standard_Real theVMax)
  : myUmin(theUMin), myVmin(theVMin), myUmax(theUMax), myVmax(theVMax), myN(0), myTolu(0.), myTolv(0.)
  {
    // CSLib_Class2d::Init
    if ((theUMax <= theUMin) || (theVMax <= theVMin) || (thePnts.Length() < 3))
      return;
    const Standard_Real aPrc = 1.e-10;
    myN = thePnts.Length();
    myTolu = theTolU; myTolv = theTolV;
    const Standard_Real du = theUMax - theUMin, dv = theVMax - theVMin;
    myX.resize(myN + 1); myY.resize(myN + 1);
    const Standard_Integer aLower = thePnts.Lower();
    for (Standard_Integer i = 0; i < myN; ++i)
    {
      const gp_Pnt2d& aP = thePnts(i + aLower);
      myX[i] = transform2d(aP.X(), theUMin, du);
      myY[i] = transform2d(aP.Y(), theVMin, dv);
    }
    myX[myN] = myX[0]; myY[myN] = myY[0];
    if (du > aPrc) myTolu /= du;
    if (dv > aPrc) myTolv /= dv;
    build();
  }

  //! Same contract as CSLib_Class2d::SiDans: 1 inside, 0 on, -1 outside.
  Standard_Integer SiDans(const gp_Pnt2d& theP) const
  {
    if (!myN)
      return 0;
    Standard_Real x = theP.X(), y = theP.Y();
    const Standard_Real aTolu = myTolu * (myUmax - myUmin), aTolv = myTolv * (myVmax - myVmin);
    if (myUmin < myUmax && myVmin < myVmax)
    {
      if ((x < (myUmin - aTolu)) || (x > (myUmax + aTolu)) || (y < (myVmin - aTolv)) || (y > (myVmax + aTolv)))
        return -1;
      x = transform2d(x, myUmin, myUmax - myUmin);
      y = transform2d(y, myVmin, myVmax - myVmin);
    }
    const Standard_Integer res = siDansOuOn(x, y);
    if (res == -1)
      return 0;
    if (myTolu || myTolv)
    {
      if (res != parity(x - myTolu, y - myTolv)) return 0;
      if (res != parity(x + myTolu, y - myTolv)) return 0;
      if (res != parity(x - myTolu, y + myTolv)) return 0;
      if (res != parity(x + myTolu, y + myTolv)) return 0;
    }
    return res ? 1 : -1;
  }

private:
  static Standard_Real transform2d(const Standard_Real u, const Standard_Real umin, const Standard_Real d)
  {
    return d > 1e-10 ? (u - umin) / d : u;   // == static Transform2d in CSLib_Class2d.cxx
  }

  Standard_Integer bucketOf(const Standard_Real v, const Standard_Real lo, const Standard_Real inv) const
  {
    const Standard_Real f = (v - lo) * inv;
    if (!(f >= 0.))
      return 0;          // also NaN
    const Standard_Integer k = (Standard_Integer)f;
    return k >= myK ? myK - 1 : k;
  }

  void build()
  {
    Standard_Real xhi = -1e300, yhi = -1e300;
    myXlo = myYlo = 1e300;
    for (Standard_Integer i = 0; i <= myN; ++i)
    {
      myXlo = std::min(myXlo, myX[i]); xhi = std::max(xhi, myX[i]);
      myYlo = std::min(myYlo, myY[i]); yhi = std::max(yhi, myY[i]);
    }
    myK = std::max(1, std::min(1 << 16, myN / 4));
    myXinv = (xhi > myXlo) ? myK / (xhi - myXlo) : 0.;
    myYinv = (yhi > myYlo) ? myK / (yhi - myYlo) : 0.;
    std::vector<Standard_Integer> cy(myK + 1, 0), cx(myK + 1, 0), cv(myK + 1, 0);
    for (Standard_Integer pass = 0; pass < 2; ++pass)
    {
      if (pass == 1)
      {
        for (Standard_Integer k = 0; k < myK; ++k) { cy[k + 1] += cy[k]; cx[k + 1] += cx[k]; cv[k + 1] += cv[k]; }
        myOy = cy; myOx = cx; myOv = cv;
        mySy.resize(cy[myK]); mySx.resize(cx[myK]); mySv.resize(cv[myK]);
      }
      for (Standard_Integer i = 0; i < myN; ++i)
      {
        const Standard_Integer ya = bucketOf(std::min(myY[i], myY[i + 1]), myYlo, myYinv);
        const Standard_Integer yb = bucketOf(std::max(myY[i], myY[i + 1]), myYlo, myYinv);
        const Standard_Integer xa = bucketOf(std::min(myX[i], myX[i + 1]), myXlo, myXinv);
        const Standard_Integer xb = bucketOf(std::max(myX[i], myX[i + 1]), myXlo, myXinv);
        const Standard_Integer kv = bucketOf(myY[i + 1], myYlo, myYinv);   // vertex ip1 = i + 1
        for (Standard_Integer k = ya; k <= yb; ++k) { if (pass == 0) ++cy[k + 1]; else mySy[cy[k]++] = i; }
        for (Standard_Integer k = xa; k <= xb; ++k) { if (pass == 0) ++cx[k + 1]; else mySx[cx[k]++] = i; }
        if (pass == 0) ++cv[kv + 1]; else mySv[cv[kv]++] = i + 1;
      }
    }
  }

  // == CSLib_Class2d::InternalSiDans restricted to the segments that can cross the ray
  Standard_Integer parity(const Standard_Real Px, const Standard_Real Py) const
  {
    const Standard_Integer k = bucketOf(Py, myYlo, myYinv);
    Standard_Integer nbc = 0;
    for (Standard_Integer j = myOy[k]; j < myOy[k + 1]; ++j)
    {
      const Standard_Integer i = mySy[j];
      const Standard_Real x = myX[i] - Px, y = myY[i] - Py, nx = myX[i + 1] - Px, ny = myY[i + 1] - Py;
      const Standard_Integer SH = (y < 0.) ? -1 : 1, NH = (ny < 0.) ? -1 : 1;
      if (NH != SH)
      {
        if (x > 0. && nx > 0.)
          nbc++;
        else if (x > 0.0 || nx > 0.)
        {
          if ((x - y * (nx - x) / (ny - y)) > 0.)
            nbc++;
        }
      }
    }
    return nbc & 1;
  }

  // == CSLib_Class2d::InternalSiDansOuOn: -1 if ON (either test, any segment), else the parity
  Standard_Integer siDansOuOn(const Standard_Real Px, const Standard_Real Py) const
  {
    {
      const Standard_Integer a = bucketOf(Py - myTolv, myYlo, myYinv), b = bucketOf(Py + myTolv, myYlo, myYinv);
      for (Standard_Integer k = a; k <= b; ++k)
        for (Standard_Integer j = myOv[k]; j < myOv[k + 1]; ++j)
        {
          const Standard_Integer v = mySv[j];
          const Standard_Real nx = myX[v] - Px, ny = myY[v] - Py;
          if (nx < myTolu && nx > -myTolu && ny < myTolv && ny > -myTolv)
            return -1;
        }
    }
    {
      const Standard_Integer k = bucketOf(Px, myXlo, myXinv);
      for (Standard_Integer j = myOx[k]; j < myOx[k + 1]; ++j)
      {
        const Standard_Integer ip1 = mySx[j] + 1;
        const Standard_Real nx = myX[ip1] - Px;
        const Standard_Real aDx = myX[ip1] - myX[ip1 - 1];
        if ((myX[ip1 - 1] - Px) * nx < 0.)
        {
          const Standard_Real aCurPY = myY[ip1] - (myY[ip1] - myY[ip1 - 1]) / aDx * nx;
          const Standard_Real aDeltaY = aCurPY - Py;
          if (aDeltaY >= -myTolv && aDeltaY <= myTolv)
            return -1;
        }
      }
    }
    return parity(Px, Py);
  }

  Standard_Real myUmin, myVmin, myUmax, myVmax;
  Standard_Integer myN;
  Standard_Real myTolu, myTolv;
  std::vector<Standard_Real> myX, myY;
  Standard_Integer myK = 1;
  Standard_Real myXlo = 0., myYlo = 0., myXinv = 0., myYinv = 0.;
  std::vector<Standard_Integer> myOy, myOx, myOv, mySy, mySx, mySv;
};

#endif
```

```cpp
//function : rebinCircles
//purpose  : defect 012
//=======================================================================
void BRepMesh_Delaun::rebinCircles (const Standard_Integer theNbTriangles)
{
  Bnd_Box2d aBox;
  IMeshData::IteratorOfMapOfInteger aTriIt (myMeshData->ElementsOfDomain());
  for (; aTriIt.More(); aTriIt.Next())
  {
    Standard_Integer aNodes[3];
    myMeshData->ElementNodes (myMeshData->GetElement (aTriIt.Key()), aNodes);
    for (Standard_Integer i = 0; i < 3; ++i)
      aBox.Add (gp_Pnt2d (GetVertex (aNodes[i]).Coord()));
  }
  if (aBox.IsVoid())
    return;
  aBox.Enlarge (Precision);
  Standard_Real aMinX, aMinY, aMaxX, aMaxY;
  aBox.Get (aMinX, aMinY, aMaxX, aMaxY);
  const Standard_Real aDX = Max (aMaxX - aMinX, Precision), aDY = Max (aMaxY - aMinY, Precision);
  const Standard_Real aTarget = Max (25.0, theNbTriangles / 2.0);
  Standard_Integer aCU = (Standard_Integer)Ceiling (Sqrt (aTarget * aDX / aDY));
  aCU = Max (1, Min (aCU, 1 << 20));
  Standard_Integer aCV = (Standard_Integer)Ceiling (aTarget / aCU);
  aCV = Max (1, Min (aCV, 1 << 20));
  initCirclesTool (aBox, aCU, aCV);
  for (aTriIt.Initialize (myMeshData->ElementsOfDomain()); aTriIt.More(); aTriIt.Next())
  {
    Standard_Integer aNodes[3];
    myMeshData->ElementNodes (myMeshData->GetElement (aTriIt.Key()), aNodes);
    myCircles.Bind (aTriIt.Key(), GetVertex (aNodes[0]).Coord(), GetVertex (aNodes[1]).Coord(),
                    GetVertex (aNodes[2]).Coord());
  }
}
```
